500|			for (let i = 0; i < buf.length; i += 4) {
501|				const m = Math.max(buf[i], buf[i + 1], buf[i + 2]);
502|				if (m > 0) nonzero++;
503|				if (m > maxRGB) maxRGB = m;
504|			}
505|			console.info(
506|				`[FreeFilmCut uInput] ${tag} frame=${this.belowDebugFrames} center ${w}x${h} nonzero=${nonzero} maxRGB=${maxRGB} tex=`,
507|				tex,
508|			);
509|		} catch (err) {
510|			console.warn("[FreeFilmCut uInput] readPixels failed", err);
511|		}
512|	}

514|		private bindBelowInputTexture({
515|		built,
516|		texture,
517|	}: {
518|		built: BuiltPackScene;
519|		texture: THREE.Texture;
520|	}) {
521|		const renderer = this.renderer;
522|		const names =
523|			built.belowInputUniformNames.length > 0
524|				? built.belowInputUniformNames
525|				: [...BELOW_INPUT_UNIFORM_NAMES];
526|
527|		// Do not mutate texture.flipY / colorSpace on shared GPU textures.
528|		if (renderer) {
529|			renderer.initTexture(texture);
530|		}
531|		this.lastBelowTexture = texture;
532|
533|		const applyToMaterial = (mat: THREE.Material | null | undefined) => {
534|			const shader = mat as THREE.ShaderMaterial;
535|			if (!shader?.isShaderMaterial || !shader.uniforms) return;
536|			let touched = false;
537|			for (const name of names) {
538|				if (!shader.uniforms[name]) {
539|					shader.uniforms[name] = { value: texture };
540|				} else {
541|					shader.uniforms[name].value = texture;
542|				}
543|				touched = true;
544|			}
545|			for (const name of BELOW_INPUT_UNIFORM_NAMES) {
546|				if (shader.uniforms[name]) {
547|					shader.uniforms[name].value = texture;
548|					touched = true;
549|				}
550|			}
551|			// Null sibling samplers (e.g. map) make uInput sample black on many GPUs.
552|			for (const [k, u] of Object.entries(shader.uniforms)) {
553|				if (!u || u.value) continue;
554|				if (/map|input|channel|tex|sampler/i.test(k)) {
555|					u.value = texture;
556|					touched = true;
557|				}
558|			}
559|			// CommonShader / many Editor defaults sample `map`, not `uInput`.
560|			// When the author did not supply a map (useMap === false), also feed
561|			// the below-composite into `map` so those packs show the video.
562|			const useMapU = shader.uniforms.useMap;
563|			const mapU = shader.uniforms.map;
564|			const useMapOn = !!(useMapU && useMapU.value);
565|			if (mapU && !useMapOn) {
566|				mapU.value = texture;
567|				touched = true;
568|			}
569|			if (touched) {
570|				shader.uniformsNeedUpdate = true;
571|				shader.toneMapped = false;
572|				if (shader.opacity === undefined || shader.opacity <= 0) {
573|					shader.opacity = 1;
574|				}
575|				if (shader.uniforms.opacity) {
576|					const o = shader.uniforms.opacity.value;
577|					if (typeof o === "number" && o <= 0) {
578|						shader.uniforms.opacity.value = 1;
579|					}
580|				}
581|			}
582|		};
583|
584|		built.root.traverse((obj) => {
585|			const mesh = obj as THREE.Mesh;
586|			if (!mesh.isMesh) return;
587|			const materials = Array.isArray(mesh.material)
588|				? mesh.material
589|				: [mesh.material];
590|			for (const mat of materials) applyToMaterial(mat);
591|		});
592|		// Also bind via uuid map (covers materials not currently on a mesh).
593|		if (built.materialsByUuid) {
594|			for (const mat of built.materialsByUuid.values()) {
595|				applyToMaterial(mat);
596|			}
597|		}
598|	}
