function placeRootAmongSiblings( group, root, where ) {

	if ( ! group || ! root ) return;
	if ( root.parent !== group ) group.add( root );

	const kids = group.children;
	const idx = kids.indexOf( root );
	if ( idx < 0 ) return;

	if ( where === 'front' ) {

		if ( idx !== kids.length - 1 ) {

			kids.splice( idx, 1 );
			kids.push( root );

		}

	} else if ( idx !== 0 ) {

		kids.splice( idx, 1 );
		kids.unshift( root );

	}

}
